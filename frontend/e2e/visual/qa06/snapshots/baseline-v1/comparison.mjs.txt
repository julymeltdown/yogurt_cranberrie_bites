// Reference pixels remain unchanged. Every crop keeps its original aspect ratio.
import {readFile,writeFile} from 'node:fs/promises'
import {resolve} from 'node:path'
import {CASES} from '../../reference-integrated/cases.mjs'
const out=process.argv[2]
if(!out?.includes('/docs/mcr/evidence/visual/qa06/'))throw Error('Use an owned QA06 evidence directory')
const report=JSON.parse(await readFile(resolve(out,'report.json')))
const esc=s=>String(s).replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]))
const reference=(r,w=270)=>`<div class="reference" style="width:${w}px;height:${w*r.height/r.width}px"><img alt="Unchanged supplied reference crop" src="../reference-target.png" style="width:${1536*w/r.width}px;left:${-r.x*w/r.width}px;top:${-r.y*w/r.width}px"></div>`
const cards=CASES.map(item=>{const shots=report.cases.filter(c=>c.id===item.id&&c.width===390&&c.screenshot)
 return `<article id="${item.id}"><h2>${esc(item.id)} <small>${esc(item.route)}</small></h2><div class="row"><figure><figcaption>Supplied reference · native aspect</figcaption>${reference(item.rect)}</figure>${shots.map(c=>`<figure><figcaption>Current ${c.era} · 390 × 844 · desktop</figcaption><a href="${esc(c.screenshot)}"><img class="capture" alt="${esc(item.id+' '+c.era)}" src="${esc(c.screenshot)}"></a></figure>`).join('')}</div><p>${report.cases.filter(c=>c.id===item.id).map(c=>`<a href="${esc(c.screenshot??'report.json')}">${c.width} ${c.era}</a>`).join(' · ')}</p></article>`}).join('')
const html=`<!doctype html><html lang="en"><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1"><title>QA06 reference reconciliation</title><style>body{margin:0 auto;padding:24px;max-width:1400px;background:#f5f3eb;color:#232320;font:15px system-ui}h1{font-size:26px}h2{font-size:22px}small{font-weight:400;font-size:14px}a{color:#284b6b}nav{display:flex;gap:14px;flex-wrap:wrap}.row{display:flex;gap:18px;overflow:auto;align-items:flex-start}figure{margin:0;flex:none}figcaption{max-width:270px;min-height:42px;font-size:13px}.reference{position:relative;overflow:hidden}.reference img{position:absolute;max-width:none;height:auto}.capture{width:270px;height:auto;display:block}article{padding:24px 0;border-bottom:1px solid #c9c6bb}p{line-height:1.5}.notice{padding:14px;border:1px solid #c9c6bb;background:#fff}button{padding:12px}</style><h1>QA06 · ten-screen reference reconciliation</h1><p class="notice">Desktop Chrome ${esc(report.browser)} at widths 360/390/430, height844. These captures use real UI with synthetic producers. Physical iOS Safari, Android Chrome, software keyboard and hardware safe areas remain NOT_RUN. Versioned-map source captures retain UNREVIEWED provenance and are not production admission. Images keep their native proportions; unequal reference/current heights are intentional.</p><nav>${CASES.map(i=>`<a href="#${i.id}">${i.id}</a>`).join('')}<a href="report.json">Raw report / hashes / limitations</a></nav>${cards}<article><h2>Versioned spatial-v4 source fixture</h2><p>Map7004 is synthetic. Actual revision5 reservation is not admission. Default-production rejection is recorded separately.</p><div class="row">${report.sourceCases.filter(c=>c.width===390).map(c=>`<figure><figcaption>${c.era} world</figcaption><img class="capture" src="${esc(c.worldScreenshot)}" alt="${c.era} source world"></figure><figure><figcaption>${c.era} map</figcaption><img class="capture" src="${esc(c.mapScreenshot)}" alt="${c.era} source map"></figure>`).join('')}</div></article></html>`
await writeFile(resolve(out,'comparison.html'),html)
console.log(resolve(out,'comparison.html'))
